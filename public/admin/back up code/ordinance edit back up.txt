    <script>
        document.addEventListener('DOMContentLoaded', function() {
            // Handle attachment row clicks
            const attachmentRows = document.querySelectorAll('.attachment-row');
            const modal = new bootstrap.Modal(document.getElementById('attachmentViewerModal'));
            
            attachmentRows.forEach(row => {
                row.addEventListener('click', function() {
                    const filePath = this.dataset.filePath;
                    const fileName = this.dataset.fileName;
                    const fileType = this.dataset.fileType.toLowerCase();
                    const fileSize = this.dataset.fileSize;
                    
                    // Update modal title and download button
                    document.getElementById('attachmentFileName').textContent = fileName;
                    document.getElementById('attachmentDownloadBtn').href = filePath;
                    document.getElementById('attachmentDownloadBtn').download = fileName;
                    
                    const previewDiv = document.getElementById('attachmentPreview');
                    previewDiv.innerHTML = '';
                    
                    // Handle different file types
                    const imageExtensions = ['jpg', 'jpeg', 'png', 'gif', 'bmp', 'webp'];
                    const pdfExtensions = ['pdf'];
                    
                    if (imageExtensions.includes(fileType)) {
                        // Show image preview
                        const img = document.createElement('img');
                        img.src = filePath;
                        img.style.maxWidth = '100%';
                        img.style.height = 'auto';
                        img.className = 'img-fluid rounded';
                        previewDiv.appendChild(img);
                    } else if (pdfExtensions.includes(fileType)) {
                        // Show PDF embed
                        const iframe = document.createElement('iframe');
                        iframe.src = filePath;
                        iframe.style.width = '100%';
                        iframe.style.height = '600px';
                        iframe.style.border = 'none';
                        iframe.style.borderRadius = '0.375rem';
                        previewDiv.appendChild(iframe);
                    } else {
                        // Show generic file preview
                        const fileInfo = document.createElement('div');
                        fileInfo.className = 'alert alert-info';
                        fileInfo.innerHTML = `
                            <div class="text-center">
                                <i class="fas fa-file fa-3x mb-3 text-info"></i>
                                <h5>File: ${fileName}</h5>
                                <p class="mb-1"><strong>Type:</strong> ${fileType.toUpperCase()}</p>
                                <p class="mb-3"><strong>Size:</strong> ${fileSize} MB</p>
                                <p class="text-muted">Click the Download button to view or save this file</p>
                            </div>
                        `;
                        previewDiv.appendChild(fileInfo);
                    }
                    
                    // Show modal
                    modal.show();
                });
            });
        });
    </script>
