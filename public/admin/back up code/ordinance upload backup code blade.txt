    <script>
        document.addEventListener('DOMContentLoaded', function() {
            // Handle file input changes for all 4 ordinance entries
            @for ($i = 0; $i < 4; $i++)
                const fileInput{{ $i }} = document.getElementById('fileInput{{ $i }}');
                const fileList{{ $i }} = document.getElementById('fileList{{ $i }}');

                fileInput{{ $i }}.addEventListener('change', function() {
                    displayFiles{{ $i }}();
                });

                function displayFiles{{ $i }}() {
                    fileList{{ $i }}.innerHTML = '';
                    const files = fileInput{{ $i }}.files;

                    if (files.length === 0) {
                        return;
                    }

                    const fileTable = document.createElement('table');
                    fileTable.className = 'table table-sm table-hover';
                    fileTable.style.marginTop = '10px';

                    const headerRow = document.createElement('tr');
                    headerRow.innerHTML = `
                        <th style="font-size: 0.75rem;"><i class="fas fa-file pe-2"></i>File Name</th>
                        <th style="font-size: 0.75rem;">Size</th>
                        <th style="font-size: 0.75rem;">Type</th>
                    `;
                    fileTable.appendChild(headerRow);

                    for (let file of files) {
                        const row = document.createElement('tr');
                        const fileSize = (file.size / 1024 / 1024).toFixed(2); // Convert to MB
                        const fileType = file.type || 'Unknown';

                        // Make rows clickable
                        row.style.cursor = 'pointer';
                        row.className = 'file-preview-row';
                        row.dataset.fileName = file.name;
                        row.dataset.fileSize = fileSize;
                        row.dataset.fileType = fileType;
                        row.dataset.fileObject = ''; // Will be set via event listener

                        row.innerHTML = `
                            <td style="font-size: 0.75rem;"><i class="fas fa-check-circle text-success pe-1"></i>${file.name}</td>
                            <td style="font-size: 0.75rem;">${fileSize} MB</td>
                            <td style="font-size: 0.75rem;"><span class="badge bg-info">${fileType.split('/')[1] || 'file'}</span></td>
                        `;

                        // Store file object reference
                        row._file = file;

                        // Add click handler
                        row.addEventListener('click', function() {
                            showFilePreview(this._file);
                        });

                        fileTable.appendChild(row);
                    }

                    fileList{{ $i }}.appendChild(fileTable);
                }
            @endfor

            // Modal functionality for file preview
            function showFilePreview(file) {
                const fileExtension = file.name.split('.').pop().toLowerCase();
                const imageExtensions = ['jpg', 'jpeg', 'png', 'gif', 'bmp', 'webp'];

                const modal = document.createElement('div');
                modal.className = 'modal fade';
                modal.id = 'filePreviewModal-' + Date.now();
                modal.tabIndex = -1;
                modal.innerHTML = `
                    <div class="modal-dialog modal-lg modal-dialog-centered">
                        <div class="modal-content">
                            <div class="modal-header">
                                <h5 class="modal-title fw-bold">${file.name}</h5>
                                <button type="button" class="btn-close" data-bs-dismiss="modal"></button>
                            </div>
                            <div class="modal-body" style="max-height: 70vh; overflow-y: auto;">
                                <div id="filePreviewContent"></div>
                            </div>
                            <div class="modal-footer">
                                <button type="button" class="btn btn-secondary" data-bs-dismiss="modal">Close</button>
                            </div>
                        </div>
                    </div>
                `;

                document.body.appendChild(modal);
                const bootstrapModal = new bootstrap.Modal(modal);

                const previewDiv = modal.querySelector('#filePreviewContent');

                if (imageExtensions.includes(fileExtension)) {
                    // Show image preview using FileReader
                    const reader = new FileReader();
                    reader.onload = function(e) {
                        const img = document.createElement('img');
                        img.src = e.target.result;
                        img.style.maxWidth = '100%';
                        img.style.height = 'auto';
                        img.className = 'img-fluid rounded';
                        previewDiv.appendChild(img);
                    };
                    reader.readAsDataURL(file);
                } else {
                    // Show file info for non-image files
                    const fileInfo = document.createElement('div');
                    fileInfo.className = 'alert alert-info';
                    fileInfo.innerHTML = `
                        <div class="text-center">
                            <i class="fas fa-file fa-3x mb-3 text-info"></i>
                            <h5>${file.name}</h5>
                            <p class="mb-1"><strong>Type:</strong> ${fileExtension.toUpperCase()}</p>
                            <p class="mb-3"><strong>Size:</strong> ${(file.size / 1024 / 1024).toFixed(2)} MB</p>
                            <p class="text-muted">This file will be uploaded with your ordinance</p>
                        </div>
                    `;
                    previewDiv.appendChild(fileInfo);
                }

                bootstrapModal.show();

                // Clean up modal from DOM when hidden
                modal.addEventListener('hidden.bs.modal', function() {
                    modal.remove();
                });
            }
        });
    </script>
